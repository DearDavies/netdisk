/*

    int order_type;
    char paras[1024];
} order_t;

int main() {
    char user_order[2048] = {0};
    while (1) {
        // 如果没有登录
        if (no_login) {
            printf("1. 注册；2. 登录；3. 退出\n");
            scanf("%d", &choice);
            if (choice == 注册) {
                // 执行注册流程，打印提示
                // 接收输入的用户名、密码
                // 将信息组织好、并发送到服务器

                // 如果服务器回复成功
                if (服务器回复注册成功) {
                    // 打印注册成功
                    // 执行登录流程
                    choice = 登录;
                }else {
                    // 提示注册失败，并重新回到初始流程
                    continue;
                }
            }
            if (choice == 登录) {
                // 执行登录流程，打印提示
                // 接收输入的用户名、密码
                // 将信息组织好、并发送到服务器

                // 如果服务器回复成功
                if (服务器回复登录成功) {
                    // 打印登录成功

                    // 修改存储状态
                    登录状态 = 已登录;
                    登录的用户名 = 用户名;
                    当前工作目录 = "/";
                }else {
                    // 提示注册失败，并重新回到初始流程
                    continue;
                }
            }
            if (choice == 退出) {
                printf("谢谢使用，再见。\n");
                return 0;
